\begin{frame}{Research objectives and overview}
\protect\hypertarget{research-objectives-and-overview}{}
We model the \textbf{dynamics of the oral-biofilm community} relevant to
peri-implantitis (SIIRI consortium).

\begin{itemize}
\tightlist
\item
  The commensal \(\to\) dysbiotic transition is described as the time
  evolution of community composition.
\item
  Central thesis: \textbf{metabolism (computed from genomes) constrains
  the sign of ecological interactions.}
\end{itemize}

\vspace{0.4em}

This deck surveys the three loosely-coupled modelling pillars and the
shared data/model contract that connects them.
\end{frame}

\begin{frame}{Graphical abstract}
\protect\hypertarget{graphical-abstract}{}
\includegraphics[width=\textwidth,height=0.66\textheight]{results/figures/concept_overview_pub.png}

Metabolism fixes the \textbf{sign} of interactions; dysbiosis appears as
a \textbf{re-wiring + spatial re-organisation} (P. gingivalis
centralises and sinks deep).
\end{frame}

\begin{frame}[fragile]{The unifying object --- the 10-guild taxonomy}
\protect\hypertarget{the-unifying-object-the-10-guild-taxonomy}{}
The canonical object shared by all data and all models is the
\textbf{class-level 10-guild taxonomy} (\texttt{GUILD\_ORDER}).

\begin{itemize}
\tightlist
\item
  \texttt{.npy} arrays and JSON files are \textbf{positionally indexed}
  by this ordering.
\item
  Each guild has one representative AGORA model (e.g.~Bacilli = \emph{S.
  gordonii}, Negativicutes = \emph{V. parvula}, Bacteroidia = \emph{P.
  melaninogenica}, Fusobacteriia = \emph{F. nucleatum}).
\item
  16S composition \(\varphi\), interaction matrix
  \(A\in\mathbb{R}^{10\times10}\), sign prior
  \(P\in\{-1,0,+1\}^{10\times10}\) --- all aligned on the same 10 axes.
\end{itemize}

\vspace{0.3em}

This common taxonomy lets the ecological, metabolic, and preprocessing
pillars speak one language.
\end{frame}

\begin{frame}[fragile]{Three loosely-coupled modelling pillars}
\protect\hypertarget{three-loosely-coupled-modelling-pillars}{}
\begin{enumerate}
\tightlist
\item
  \textbf{Ecological ODE inference (the paper)} --- infer a gLV /
  Hamilton interaction matrix from longitudinal 16S, regularise with a
  \textbf{metabolic sign prior}, validate by leave-one-patient-out CV
  (LOO-CV).
\item
  \textbf{COMETS / dFBA} --- mechanistic genome-scale (AGORA) community
  simulation, cross-validating the inferred interactions independently.
\item
  \textbf{16S / metadata preprocessing} --- raw sequencing + paper
  supplements \(\to\) guild abundance arrays.
\end{enumerate}

\vspace{0.4em}

The three run independently but are connected by the \textbf{10-guild
taxonomy} and the \textbf{\texttt{fit\_*.json} interchange format.}
\end{frame}

\begin{frame}{Core model (concise)}
\protect\hypertarget{core-model-concise}{}
Generalized Lotka--Volterra for absolute abundances \(x_i\):
\[\dot{x}_i = x_i\Big(b_i + \sum_{j=1}^{S} A_{ij}\,x_j\Big),\qquad S=10 .\]

Replicator / Hamilton form on the simplex \(\sum_i\varphi_i=1\) (16S is
compositional):
\[\dot{\varphi}_i = \varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big].\]

\textbf{Sign prior:} from AGORA cross-feeding flux \(F_{ij}\) build
\[P_{ij}=\operatorname{sgn}(F_{ij})\in\{-1,0,+1\},\] constraining only
the \textbf{sign} of \(A\) (magnitude discarded by design).
\end{frame}

\begin{frame}{Spatial extension (Heine HOBIC FISH)}
\protect\hypertarget{spatial-extension-heine-hobic-fish}{}

Depth-resolved 5-species profiles from CLSM-FISH are fit by a
reaction--diffusion equation. Dysbiosis is not merely a compositional
change but a \textbf{spatial reorganisation}: \emph{P. gingivalis} sinks

\end{frame}

\begin{frame}[fragile]{How the sub-decks connect}
\protect\hypertarget{how-the-sub-decks-connect}{}
\begin{itemize}
\tightlist
\item
  \textbf{AGORA} (sign prior): the metabolic bridge building
  \(\operatorname{sgn}(F_{ij})\to P_{ij}\).
\item
  \textbf{Dieckow (deck A):} infers \(A\) under that prior and validates
  by LOO.
\item
  \textbf{Network (deck B):} analyses the structure of inferred \(A\)
  --- centrality, concordant backbone, CS\(\leftrightarrow\)DH rewiring.
\item
  \textbf{Spatial PDE (deck C):} extends the temporal dynamics to
  depth-resolved reaction--diffusion.
\item
  \textbf{FISH pipeline:} CLSM \texttt{.lif} \(\to\) depth profiles (the
  PDE observable).
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Take-home}
\protect\hypertarget{take-home}{}
\begin{enumerate}
\tightlist
\item
  \textbf{Metabolism constrains the sign of ecological interactions}
  (\(P_{ij}=\operatorname{sgn}(F_{ij})\); magnitude is not used).
\item
  Dysbiosis is a \textbf{rewiring + spatial reorganisation}, with
  \emph{P. gingivalis} sinking into the deep anaerobic niche.
\item
  That claim reproduces across \textbf{two cohorts} (Dieckow \(\times\)
  Botelho, 89\%, \(p\approx0.02\)) and an \textbf{independent
  mechanistic route} (COMETS dFBA).
\item
  One \textbf{10-guild taxonomy} and the \textbf{\texttt{fit\_*.json}}
  format tie it all together.
\end{enumerate}
\end{frame}

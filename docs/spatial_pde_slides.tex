\begin{frame}[fragile]{支配方程式（反応拡散 PDE）}
\protect\hypertarget{ux652fux914dux65b9ux7a0bux5f0fux53cdux5fdcux62e1ux6563-pde}{}
深さ \(z\in[0,L]\)、組成 \(\varphi_i(z,t)\) に対する反応 --- 移流 ---
拡散方程式： \[\frac{\partial \varphi_i}{\partial t}
 = D_i\,\frac{\partial^2 \varphi_i}{\partial z^2}
 \;-\; u\,\frac{\partial \varphi_i}{\partial z}
 \;+\; R_i(\varphi).\]

\begin{itemize}
\tightlist
\item
  \(R_i(\varphi)\) は replicator / Hamilton 反応項。\(A,b\) は \textbf{①
  Heine 5 菌種 TMCMC posterior}（\(N_p=10{,}000\),
  \texttt{ultimate\_10000p}）から
  \textbf{固定}（本デッキはその空間拡張）。
  \[R_i(\varphi)=\varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big]+\gamma\,\varphi_i.\]
\item
  \textbf{自由パラメータは拡散率 \(D_i\) と移流 \(u\)
  のみ}（反応は①で確定済み）。
\item
  \(\gamma\) は Lagrange 項：各 \(z\) で
  \(\sum_i \varphi_i + \varphi_0 = 1\) を強制する。
\end{itemize}
\end{frame}

\begin{frame}{境界条件と数値解法}
\protect\hypertarget{ux5883ux754cux6761ux4ef6ux3068ux6570ux5024ux89e3ux6cd5}{}
\textbf{境界条件：}
\[\left.\frac{\partial \varphi_i}{\partial z}\right|_{z=0}=0\quad(\text{基質，no-flux}),\qquad
\varphi_i\big|_{z=L}=\varphi_{\text{bulk},i}\quad(\text{bulk，Dirichlet}).\]
\(\varphi_{\text{bulk},i}\) は Day 1 の中央値で固定。

\textbf{数値解法（Method of Lines）：} Lie
演算子分割で反応と輸送を交互に解く。
\[e^{\Delta t\,\mathcal{L}} \approx e^{\Delta t\,\mathcal{R}}\,e^{\Delta t\,\mathcal{T}} + \mathcal{O}(\Delta t).\]

\begin{itemize}
\tightlist
\item
  反応 \(\mathcal{R}\)：各 \(z\) ノードで陰的 Euler（Newton 反復、stiff
  対応）。
\item
  輸送 \(\mathcal{T}\)：陽的有限差分（中心差分 \(\partial_z^2\)、風上
  \(\partial_z\)）。
\end{itemize}
\end{frame}

\begin{frame}{逆問題}
\protect\hypertarget{ux9006ux554fux984c}{}
観測 \(\varphi^{\text{obs}}(z,t)\)
に対する輸送パラメータの最小二乗推定： \[\min_{D,\,u}\;\; \sum_t \sum_z
 \big\lVert \varphi^{\text{pred}}(z,t;D,u) - \varphi^{\text{obs}}(z,t)\big\rVert^2 .\]

\begin{itemize}
\tightlist
\item
  最適化は \textbf{L-BFGS-B}。stiff PDE を通した数値勾配（JAX
  前向きモデル）。
\item
  前向きモデルは上記 MoL ソルバ。\(A,b\) 固定なので未知は
  \(D\in\mathbb{R}^5\) と \(u\)。
\item
  無次元化：\textbf{Péclet 数} \(\mathrm{Pe} = uL/D_i\) が移流 vs
  拡散の優劣を表す。 \(\mathrm{Pe}\gg1\) で移流支配、\(\mathrm{Pe}\ll1\)
  で拡散支配。
\end{itemize}
\end{frame}

\begin{frame}{その他：bulk は CH ≈ DH}
\protect\hypertarget{ux305dux306eux4ed6bulk-ux306f-ch-dh}{}
CH--DH の \textbf{Bray--Curtis divergence} \(\approx 0.2\)、Day 1
から\textbf{ほぼ一定}。

\begin{itemize}
\tightlist
\item
  差は接種（inoculum）で決まり、時間発展でほとんど拡大しない。
\item
  bulk 組成では CH \(\approx\) DH \(\Rightarrow\) dysbiosis は
  \textbf{空間的・時間的}であり、 bulk 組成のシフトではない。
\end{itemize}

\vspace{0.4em}

これは知見 1・2 と整合する：違いは「誰がどれだけいるか」ではなく
「どこにいて、何と組むか」に現れる。
\end{frame}

\begin{frame}{拡散率 vs 生態的中心性}
\protect\hypertarget{ux62e1ux6563ux7387-vs-ux751fux614bux7684ux4e2dux5fc3ux6027}{}
\includegraphics[width=\textwidth,height=0.52\textheight]{results/diffusion_fit/d_vs_centrality.png}

フィットした拡散率 \(D_i\) は gLV 相互作用強度（中心性）と
\textbf{負相関}： CH \(r=-0.40\)、DH \(r=-0.47\)（5 種、illustrative）。

\vspace{0.2em}

\(\Rightarrow\)
「中心的な種ほど動かない」。生態的ハブが空間的に固定される傾向。
\end{frame}

\begin{frame}{in-vitro ↔ in-vivo マッピング}
\protect\hypertarget{in-vitro-in-vivo-ux30deux30c3ux30d4ux30f3ux30b0}{}
\includegraphics[width=\textwidth,height=0.5\textheight]{results/diffusion_fit/hobic_vs_dieckow.png}

HOBIC の 5 種を Dieckow の 5 ギルドに対応づけ、\textbf{順位が一致}：
Spearman \(\rho = 0.70\)。

\begin{itemize}
\tightlist
\item
  in-vivo は \emph{S. oralis} 優占、定義済み inoculum は均等接種。
\item
  順位構造は保たれる \(\Rightarrow\) in-vitro モデルは in-vivo
  の縮約として妥当。
\end{itemize}
\end{frame}

\begin{frame}{結論}
\protect\hypertarget{ux7d50ux8ad6}{}
\begin{enumerate}
\tightlist
\item
  bulk ODE（WHO）に FISH 深さプロファイル（WHERE）を足し、反応項固定で
  \textbf{輸送パラメータ \(D_i,u\) のみ}を逆推定する枠組みを構築。
\item
  \textbf{dysbiosis \(=\) 空間的再編成}：\emph{P. gingivalis}
  が深部へ沈み（\(+30\,\mu\mathrm{m}\)）、 \emph{F. nucleatum}
  から自律化（Manders \(M_1\) が初期のみ高い）。
\item
  bulk は CH \(\approx\) DH（Bray--Curtis \(\approx0.2\) で一定）---
  違いは bulk でなく空間。
\item
  「中心的な種ほど動かない」（\(D\) と中心性が負相関）；in-vitro↔in-vivo
  の 順位は一致（\(\rho=0.70\)）。
\item
  \textcolor{red}{輸送パラメータは**暫定**（未収束）}。HPC で
  sweep・高速収束設定が収束中。
\end{enumerate}
\end{frame}

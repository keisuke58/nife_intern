\begin{frame}{Motivation --- peri-implantitis and biofilm dysbiosis}
\protect\hypertarget{motivation-peri-implantitis-and-biofilm-dysbiosis}{}
Peri-implantitis is driven by a shift of the peri-implant oral biofilm
from a \textbf{commensal} to a \textbf{dysbiotic} multi-species
community.

\begin{itemize}
\tightlist
\item
  The ecological driver is the \textbf{species--species interaction
  matrix} \(A\).
\item
  \textbf{Goal:} infer \(A\) from compositional time-series, and
  quantify how the health \(\to\) disease transition reshapes it.
\item
  \textbf{Thesis.} All 15 interaction parameters are recovered from data
  alone (no a-priori sign locking); the health/disease difference is a
  \textbf{topological reorganisation}, not a simple rescaling.
\end{itemize}
\end{frame}

\begin{frame}{Data --- Heine et al.~(2025), four conditions}
\protect\hypertarget{data-heine-et-al.-2025-four-conditions}{}
Longitudinal relative-abundance time-series of a \textbf{five-species}
oral biofilm, under \textbf{four conditions}:

\begin{itemize}
\tightlist
\item
  \textbf{Commensal} vs \textbf{dysbiotic} community \(\times\)
  \textbf{static} vs \textbf{HOBIC} reactor flow \(\Rightarrow\) CS, CH,
  DS, DH.
\item
  Five species: \emph{S. oralis} (So), \emph{A. naeslundii} (An),
  \emph{Veillonella} (Vei), \emph{F. nucleatum} (Fn), \emph{P.
  gingivalis} (Pg).
\item
  Each sample is a composition \(\varphi\in\Delta^{4}\) with
  \(\sum_i\varphi_i=1\).
\end{itemize}
\end{frame}

\begin{frame}{Model --- continuum biofilm / replicator dynamics}
\protect\hypertarget{model-continuum-biofilm-replicator-dynamics}{}
State: relative abundances \(\varphi_i\) on the simplex. The Extended
Hamilton Principle yields replicator dynamics with a \textbf{symmetric}
interaction matrix \(A\):
\[\dot{\varphi}_i = \varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big],
\qquad \sum_i\varphi_i = 1 .\]

\begin{itemize}
\tightlist
\item
  Symmetric \(A\) (5\$\times\$5) \(\Rightarrow\) \textbf{15 free
  parameters}.
\item
  \(A_{ij}>0\): facilitation; \(A_{ij}<0\): suppression / competition.
\end{itemize}
\end{frame}

\begin{frame}{Inference --- Bayesian framework}
\protect\hypertarget{inference-bayesian-framework}{}
\[\mathcal{L}(A,b)=\frac{1}{2\sigma^{2}}\sum_{t}\big\lVert\varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}\big\rVert^{2}
\;+\; \text{(weakly-informative prior)} .\]

\begin{itemize}
\tightlist
\item
  \textbf{All 15 entries free} --- no a-priori sign constraint, so any
  sparsity is data-driven.
\item
  Posterior sampled by \textbf{Transitional MCMC (TMCMC)} with tempered
  likelihood and \(N_p\) particles.
\end{itemize}
\end{frame}

\begin{frame}{Two-phase estimation strategy}
\protect\hypertarget{two-phase-estimation-strategy}{}
\begin{itemize}
\tightlist
\item
  \textbf{Phase 1} (fix \(\psi\), \(N_p=1{,}000\)): rapid, robust
  identification of \(A\).
\item
  \textbf{Phase 2} (free \(\psi\), \(N_p=10{,}000\)): full joint
  posterior, physically consistent.
\end{itemize}

Phase-1 MAP is already close to Phase-2
(\(\Delta\mathrm{RMSE}\le 0.03\)), confirming the fix-\(\psi\) stage is
a robust warm start.
\end{frame}

\begin{frame}[fragile]{GPU acceleration}
\protect\hypertarget{gpu-acceleration}{}
\begin{itemize}
\tightlist
\item
  \textbf{JAX + \texttt{vmap}} on a single RTX 4090.
\item
  \textbf{≈200× speedup} vs CPU; \(N_p=10{,}000\) in ≈2 h per condition.
\item
  Makes production runs of the full \textbf{15-D posterior} feasible.
\end{itemize}
\end{frame}

\begin{frame}{Discussion --- biological interpretation}
\protect\hypertarget{discussion-biological-interpretation}{}
\textbf{Commensal state (CS, CH)}

\begin{itemize}
\tightlist
\item
  So--An co-aggregation + So--Vei lactate exchange: a stable
  \textbf{mutualistic core}.
\item
  Fn, Pg entries emerge near zero --- data-driven sparsity.
\end{itemize}

\textbf{Dysbiotic transition (DS, DH)}

\begin{itemize}
\tightlist
\item
  \textbf{Vei\(\to\)Pg} (pH) and \textbf{Fn\(\to\)Pg} (peptides)
  strongly activated by W83.
\item
  \(\rho(\mathrm{DH}-\mathrm{CS})=-0.49\): \textbf{topological
  reorganisation}, not recoverable by parameter rescaling.
\end{itemize}
\end{frame}

\begin{frame}{Conclusions}
\protect\hypertarget{conclusions}{}
\begin{enumerate}
\tightlist
\item
  \textbf{Two-phase full-discovery TMCMC:} Phase 1 (fix-\(\psi\)) gives
  rapid robust \(A\); Phase 2 (free \(\psi\)) gives a consistent joint
  posterior. All 15 parameters free --- \textbf{emergent sparsity} from
  data.
\item
  \textbf{GPU-accelerated TMCMC:} ≈200× speedup; \(N_p=10{,}000\) in ≈2
  h per condition.
\item
  \textbf{Validated interpretation:} pH \(r=0.84\), gingipain
  \(r=0.90\); dysbiosis is a \textbf{topological reorganisation}
  (\(\rho=-0.49\)).
\end{enumerate}
\end{frame}

\begin{frame}{生態モデル}
\protect\hypertarget{ux751fux614bux30e2ux30c7ux30eb}{}
絶対量 \(x_i\) の一般化 Lotka--Volterra：
\[\dot{x}_i = x_i\Big(b_i + \sum_{j=1}^{S} A_{ij}\,x_j\Big),\qquad i=1,\dots,S=10 .\]

単体上の組成 \(\varphi_i = x_i/\sum_k x_k\)（16S は組成データ）に対する
replicator / Hamilton 形：
\[\dot{\varphi}_i = \varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big],
\qquad \sum_i \varphi_i = 1 .\]

\begin{itemize}
\tightlist
\item
  \(A_{ij}>0\)：\(j\) が \(i\)
  を\textbf{促進}、\(A_{ij}<0\)：\textbf{抑制}。
\item
  推定は\textbf{劣決定}：\(\mathcal{O}(S^2)=100\) パラメータ vs \(10\)
  患者 \(\times\,3\) 週。 存在量だけでは \(A\) の符号は識別不能
  \(\Rightarrow\) \textbf{代謝符号 prior} を加える。
\end{itemize}
\end{frame}

\begin{frame}{FBA（Flux Balance Analysis）= 線形計画}
\protect\hypertarget{fbaflux-balance-analysis-ux7ddaux5f62ux8a08ux753b}{}
ゲノムスケールモデルは化学量論行列 \(S\in\mathbb{R}^{m\times n}\)
（代謝物 \(\times\) 反応）を与える。定常成長は LP：
\[\max_{v}\; c^{\!\top} v \quad\text{s.t.}\quad S v = 0,\;\; v^{-}\le v \le v^{+},\]
\(c\) がバイオマス反応を選ぶ（\(\mu = c^{\!\top}v\)）。\textbf{pFBA}
はフラックスループを 第2段で除去：
\[\min_{v}\; \sum_{j}\lvert v_j\rvert \quad\text{s.t.}\quad c^{\!\top}v=\mu^{\star},\; Sv=0,\; v^{-}\le v\le v^{+}.\]

交換フラックスから、ギルドごとの \textbf{分泌}
\(s_{j\alpha}=\left[v^{\text{ex}}_{j\alpha}\right]_{+}\)、
\textbf{取り込み}
\(u_{i\alpha}=\left[-v^{\text{ex}}_{i\alpha}\right]_{+}\)（代謝物
\(\alpha\)）を得る。
\end{frame}

\begin{frame}{AGORA2 が与えるもの}
\protect\hypertarget{agora2-ux304cux4e0eux3048ux308bux3082ux306e}{}
\textbf{AGORA2}（Heinken et al., \emph{Nat. Biotechnol.} 2023,
41:1320）：\textbf{7,302} 株の
ゲノムスケール再構成。腸内から\textbf{口腔}菌へ拡張。ギルドごとに代表
SBML を 1 つ。

\vspace{0.5em}
\begin{itemize}
\tightlist
\item
  代表株：Bacilli = \emph{S. gordonii}、Negativicutes = \emph{V.
  parvula}、 Bacteroidia = \emph{P. melaninogenica}、Fusobacteriia =
  \emph{F. nucleatum} \ldots{}
\item
  口腔液培地（Dawes
  2008）：糖・20アミノ酸・B群ビタミン・微量元素・細胞壁前駆体
  \(\Rightarrow\) 全 10
  ギルドで正の増殖（\(\mu = 0.11\text{–}1.66\,\mathrm{h^{-1}}\)）。
\end{itemize}

\vspace{0.5em}

培地組成は重要な感度因子であり、栄養が乏しい場合は \(\mu\to0\) となり
cross-feeding シグナルが消失する。
\end{frame}

\begin{frame}{cross-feeding スコア \(\to\) 符号 prior}
\protect\hypertarget{cross-feeding-ux30b9ux30b3ux30a2-to-ux7b26ux53f7-prior}{}
毒素を \(\mathcal{T}=\{\text{H}_2\text{O}_2,\text{H}_2\text{S}\}\)
とし、\(j\to i\) の \textbf{正味代謝フロー}を定義：
\[F_{ij} \;=\; \underbrace{\sum_{\alpha\notin\mathcal{T}} w_\alpha\, s_{j\alpha}\,u_{i\alpha}}_{\text{cross-feeding }(+)}
\;-\; \underbrace{\sum_{\alpha\in\mathcal{T}} w_\alpha\, s_{j\alpha}\,u_{i\alpha}}_{\text{toxin }(-)} .\]

\textbf{符号 prior} は方向のみを残す：
\[P_{ij} = \operatorname{sgn}(F_{ij}) \in \{-1,\,0,\,+1\}.\]

大きさは設計上捨てる --- FBA フラックス単位 \(\neq\) 生態的単位。
\(10\times 9=90\) 方向ペアから制約集合
\(\mathcal{M}=\{(i,j): P_{ij}\neq 0\}\) を得る。
\end{frame}

\begin{frame}{ベイズ統合：符号 prior の罰則}
\protect\hypertarget{ux30d9ux30a4ux30baux7d71ux5408ux7b26ux53f7-prior-ux306eux7f70ux5247}{}
フィットは軌道誤差に、\textbf{符号違反のみ}（大きさは不問）を罰する片側ヒンジを加える：
\[\mathcal{L}(A,b) = \frac{1}{2\sigma^{2}}\sum_{t}\big\lVert \varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}(A,b)\big\rVert^{2}
\;+\; W\!\!\sum_{(i,j)\in\mathcal{M}}\!\! \left[-\,P_{ij}\,A_{ij}\right]_{+} .\]

\begin{itemize}
\tightlist
\item
  \(\left[-P_{ij}A_{ij}\right]_{+}=\max(0,-P_{ij}A_{ij})\) は
  \(\operatorname{sgn}(A_{ij})=P_{ij}\) のとき \(0\)。
\item
  \(W\) = prior の硬さ、\(\sigma\) = 観測スケール。
\item
  5 種アトラクターの posterior は \textbf{TMCMC}（\(10^4\)
  粒子）、ギルドは L-BFGS-B / TMCMC。
\end{itemize}
\end{frame}

\begin{frame}{なぜ量 prior は失敗するか --- MacArthur の視点}
\protect\hypertarget{ux306aux305cux91cf-prior-ux306fux5931ux6557ux3059ux308bux304b-macarthur-ux306eux8996ux70b9}{}
消費者--資源理論（MacArthur 1970；Marsland 2019）は次を導く：
\[A_{ij} = \underbrace{\sum_{\alpha} s_{j\alpha}\,c_{i\alpha}}_{\text{cross-feeding}\,(+)}
\;-\; \underbrace{\frac{c_i\!\cdot\! c_j}{\lVert c_i\rVert\,\lVert c_j\rVert}}_{\text{niche overlap}\,(-)} .\]

\vspace{0.5em}
\begin{itemize}
\tightlist
\item
  \textbf{ニッチ重複（cosine）項が飽和}：口腔菌は generalist で
  \(\cos(c_i,c_j)\approx1\) → ほぼ全ペアが競争と誤判定（実測 正 \(6\) /
  負 \(84\) ペア＝使えない）。
\item
  Growth-rate-suppression も貧栄養培地で同様に破綻。
\end{itemize}

\vspace{0.5em}
\(\Rightarrow\) \(\operatorname{sgn}(A_{ij})\) を残し、大きさは捨てる。
\end{frame}

\begin{frame}{MICOM --- 群集 FBA}
\protect\hypertarget{micom-ux7fa4ux96c6-fba}{}
単種 pFBA は\textbf{実行可能性}のみ（\(j\) が \(X\) を出せる・\(i\) が
\(X\) を食べられる）。 \textbf{MICOM}（Diener
2020）は群集を同時に解き、実際にフラックスが\textbf{流れるか}を、
\textbf{cooperative trade-off} で判定する：
\[\max\; \min_{i}\frac{\mu_i}{\mu_i^{\max}}\quad\text{s.t.}\quad
\sum_i \mu_i \ge \tau\sum_i \mu_i^{\max},\;\; S^{\text{com}}v=0,\;\tau=0.5 .\]

\vspace{0.5em}
\begin{itemize}
\tightlist
\item
  各菌に最大成長の \(\tau\)
  倍以上を保証した上で、共有フラックス配分を確定。
\item
  generalist でも\textbf{実際に群集フラックスを運ぶ経路のみ}が活性化 →
  「全員競合」の 非特異アーティファクトが消える。
\item
  群集解の交換フラックス \(s_{j\alpha},u_{i\alpha}\) をそのまま
  \(F_{ij}\) に渡す。
\end{itemize}
\end{frame}

\begin{frame}{限界と結論}
\protect\hypertarget{ux9650ux754cux3068ux7d50ux8ad6}{}
\textit{ギルド $=$ class（代表株 $\neq$ ギルド全体）；阻害行の $20/22$ は酸素（生産者不在）$\Rightarrow$ 毒素は H$_2$O$_2$ のみ発火；$w_\alpha$ は代謝物ごと max；大きさは捨てている。}

\vspace{0.4em}
\textbf{結論：}
\begin{enumerate}
\tightlist
\item \(\text{AGORA pFBA}\to F_{ij}\to P_{ij}=\operatorname{sgn}(F_{ij})\) という変換が方法論的新規性。
\item \textbf{符号は使え、量は使えない}（MacArthur cosine 飽和を回避）。
\item 単種（\(92\%\)）\(\to\) \textbf{MICOM（\(100\%\)）} が群集文脈を捕捉。
\item 誠実な検証：\textbf{協調方向のみ} \(p=4\times10^{-4}\)；2 コホート \(89\%\)；prior はモデリング選択。

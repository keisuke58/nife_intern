\begin{frame}{Scope}
\protect\hypertarget{scope}{}
From \textbf{compositional 16S time-series} of an oral biofilm, infer a
\textbf{signed interaction matrix} \(A\), and quantify how far that
inference can be trusted.

\begin{itemize}
\tightlist
\item
  model: generalized Lotka--Volterra (gLV) / Hamilton replicator on the
  simplex
\item
  constraint: an AGORA2-derived \textbf{metabolic sign prior} (see the
  AGORA deck; cited here only briefly)
\item
  focus: the \textbf{fit}, \textbf{LOO-CV}, model comparison, and
  \textbf{critical validation}
\end{itemize}

\vspace{0.4em}

\textbf{Claim.} The prior provides sign-consistency and interpretability
rather than predictive accuracy. What is validated independently of the
data is the \textbf{cross-feeding direction only} (below).
\end{frame}

\begin{frame}[fragile]{Data and design}
\protect\hypertarget{data-and-design}{}
Dieckow et al.~2024 (ENA \textbf{PRJEB71108}): a longitudinal 16S series
of \textbf{10 patients \(\times\) 3 weeks}, aggregated to \textbf{10
class-level guilds} (\texttt{GUILD\_ORDER} is canonical).

\begin{center}\includegraphics[width=0.86\textwidth,keepaspectratio]{dieckow_paper/figures/fig0_study_design.png}\end{center}

Early peri-implant colonization. Each sample is treated as a composition
\(\varphi\in\Delta^{9}\) with \(\sum_i\varphi_i=1\).
\end{frame}

\begin{frame}{Model --- gLV and replicator}
\protect\hypertarget{model-glv-and-replicator}{}
Generalized Lotka--Volterra for absolute abundances \(x_i\):
\[\dot{x}_i = x_i\Big(b_i + \sum_{j=1}^{S} A_{ij}\,x_j\Big),\qquad S=10 .\]

On the simplex \(\varphi_i=x_i/\sum_k x_k\), the replicator / Hamilton
form:
\[\dot{\varphi}_i = \varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big],
\qquad \sum_i\varphi_i=1 .\]

\begin{itemize}
\tightlist
\item
  \textbf{Hamilton} uses a \textbf{symmetric} \(A\) (replicator with
  linear payoffs; Taylor \& Jonker 1978).
\item
  \textbf{Classic gLV} uses an \textbf{asymmetric} \(A\). \(A_{ij}>0\):
  facilitation; \(A_{ij}<0\): suppression.
\end{itemize}
\end{frame}

\begin{frame}{Inference is underdetermined \(\Rightarrow\) sign prior}
\protect\hypertarget{inference-is-underdetermined-rightarrow-sign-prior}{}
Parameters \(\mathcal{O}(S^2)\approx100\) vs \(10\times3\) observations.
Abundance alone cannot identify the signs of \(A\) \(\Rightarrow\) a
one-sided hinge penalises \textbf{sign violations only}:
\[\mathcal{L}(A,b)=\frac{1}{2\sigma^{2}}\sum_{t}\big\lVert\varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}\big\rVert^{2}
\;+\; W\!\!\sum_{(i,j)\in\mathcal{M}}\!\!\left[-\,P_{ij}\,A_{ij}\right]_{+} .\]

\begin{itemize}
\tightlist
\item
  \(P_{ij}=\operatorname{sgn}(F_{ij})\) comes from AGORA cross-feeding
  flux (derived in the AGORA deck) (explicit AGORA-output example on the
  next slide).
\item
  \(\left[-P_{ij}A_{ij}\right]_{+}\) is \(0\) when
  \(\operatorname{sgn}(A_{ij})=P_{ij}\) --- \textbf{magnitude is free}.
\item
  Posterior sampled by \textbf{TMCMC} (transitional MCMC): \(10^{4}\)
  \textbf{particles} (parameter draws) are advanced through tempered
  stages from prior \(\to\) posterior, reweighted and resampled each
  stage. The final particle cloud \emph{is} the posterior over
  \((A,b)\); \(10^{4}\) sets its resolution.
\end{itemize}
\end{frame}

\begin{frame}{From AGORA output to the sign prior \(P_{ij}\)}
\protect\hypertarget{from-agora-output-to-the-sign-prior-p_ij}{}
\textbf{What AGORA produces.} For each guild, an AGORA2 pFBA solution
gives a \textbf{secretion/uptake flux profile}
(mmol,gDW\(^{-1}\)h\(^{-1}\)). Example (real pFBA output):

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Guild & pFBA secretion (mmol\,gDW$^{-1}$h$^{-1}$) \\
\midrule
Actinobacteria & secretes acetate 49.8, formate 53.7, succinate 39.2 \\
Bacilli & secretes acetate 921, lactate 982, propionate 1000, succinate 18 \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Net directed flow.} Donor \(i\) secretes a metabolite that
acceptor \(j\) consumes \(\Rightarrow\) directed cross-feeding flux
\(F_{ij}>0\); competition for a shared resource
\(\Rightarrow F_{ij}<0\).

\textbf{Derivation.} \(P_{ij}=\operatorname{sgn}(F_{ij})\) --- only the
\emph{sign} enters the penalty (FBA flux units \(\neq\) ecological
units, so magnitude is discarded by design).

\textbf{Worked example.} Actinobacteria secretes
acetate/formate/succinate, which Bacilli consume
\(\Rightarrow F>0 \Rightarrow P=+1\) (facilitation), matching the fitted
\(\hat A=+1.62\). Key literature pairs confirmed 2/3; inhibitory links
(H\(_2\)O\(_2\), competition) need community-level modelling.
\end{frame}

\begin{frame}{LOO-CV definition}
\protect\hypertarget{loo-cv-definition}{}
Re-fit with patient \(p\) held out, evaluate on its trajectory:
\[\mathrm{RMSE}=\sqrt{\frac{1}{NT}\sum_{t}\big\lVert\varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}\big\rVert^{2}},\]
\[\mathrm{BC}=\frac{\sum_i\lvert\varphi^{\text{obs}}_i-\varphi^{\text{pred}}_i\rvert}{\sum_i\big(\varphi^{\text{obs}}_i+\varphi^{\text{pred}}_i\big)}
\quad(\text{Bray–Curtis dissimilarity}).\]

Every one of the 10 patients is held out in turn (leave-one-patient-out)
and averaged.
\end{frame}

\begin{frame}{Two-cohort replication (cross-design check)}
\protect\hypertarget{two-cohort-replication-cross-design-check}{}
Duran-Pinedo et al.~2021 (\textbf{BMC Biol 19:240}; ENA
\textbf{PRJNA725874}, 15 patients \(\times\) 7 timepoints) is fit
\textbf{prior-free} and independently; strong-pair signs are then
compared to Dieckow.

\begin{center}\includegraphics[height=0.36\textheight,keepaspectratio]{results/duranpinedo_validation/fig_duranpinedo_A_comparison_noprior.png}\end{center}

Strong-pair directed signs agree at \textbf{\(89\%\)} (\(8/9\)
upper-triangular, \(p\approx0.02\)).
\textcolor{red}{\textbf{Caveat --- not a like-for-like replication.}}
The cohorts differ in \textbf{timescale and clinical state}: Dieckow =
early peri-implant colonization (weeks; health \(\to\) colonization);
Duran-Pinedo = long-term \textbf{periodontitis} progression (disease).
The 89\% is a coarse \emph{strong-pair sign} agreement across designs,
not a matched replication. The Actinobacteria-axis correspondence is
\textbf{suggestive} and needs a matched-design cohort to confirm.
\end{frame}

\begin{frame}{Interpretation and limitations}
\protect\hypertarget{interpretation-and-limitations}{}
\small

\begin{itemize}
\tightlist
\item
  The \textbf{AGORA prior itself is not reproduced} by the 16S dynamics
  \(\Rightarrow\) it is a \textbf{modelling choice}, not a
  data-confirmed fact.
\item
  What is validated independently of the data:

  \begin{enumerate}
  \tightlist
  \item
    the \textbf{cross-feeding direction} (Hamilton,
    \(p=4\times10^{-4}\)),
  \item
    \textbf{two-cohort strong-pair} signs (Dieckow \(\times\)
    Duran-Pinedo, \(89\%\); cross-design --- see caveat).
  \end{enumerate}
\item
  Competition direction, prior magnitude, and the full matrix are not
  validated.
\item
  The MICOM--gLV RMSE improvement (\(p\approx0.07\), \(N=10\)) is
  \textbf{structurally underpowered} (power \(\approx0.30\) for
  \(d=0.5\)); read as a consistency result, not a powered effect. The
  permutation test (\(p=4\times10^{-4}\)) is the primary statistical
  claim.
\item
  A \textbf{neutral-initialised no-prior LOO} (zero-init, submitted)
  will confirm the enrichment is data-driven and not a warm-start
  artefact.
\end{itemize}

\vspace{0.3em}

To avoid overstating the result: a \textbf{subset} of signs is
supported; the remainder remains hypothesis.
\end{frame}

\begin{frame}{Conclusions}
\protect\hypertarget{conclusions}{}
\begin{enumerate}
\tightlist
\item
  Compositional 16S time-series \(\to\) a signed \(A\), via gLV /
  Hamilton replicator.
\item
  Under-determination is eased by the \textbf{AGORA metabolic sign
  prior} (one-sided hinge); posterior via TMCMC.
\item
  Best: L1+L2+AGORA \(W=1.0\) gives \textbf{LOO-RMSE \(0.0504\)}, SA
  \(100\%\). But prior-free gLV reaches \(0.0455\) --- the prior
  provides \textbf{interpretability} rather than accuracy.
\item
  \textbf{Signs are stable under LOO} (\(\geq0.70\)); the MacArthur
  magnitude prior fails.
\item
  Honest validation: only the \textbf{cross-feeding direction}
  (\(p=4\times10^{-4}\)) and \textbf{two-cohort strong pairs} (\(89\%\))
  are independently supported. Clinical validation is in progress.
\end{enumerate}
\end{frame}

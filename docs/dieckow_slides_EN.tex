\begin{frame}{Scope}
\protect\hypertarget{scope}{}
From \textbf{compositional 16S time-series} of an oral biofilm, infer a
\textbf{signed interaction matrix} \(A\), and quantify how far that
inference can be trusted.

\begin{itemize}
\tightlist
\item
  model: generalized Lotka--Volterra (gLV) / Hamilton replicator on the
  simplex
\item
  constraint: an AGORA2-derived \textbf{metabolic sign prior} (see the
  AGORA deck; cited here only briefly)
\item
  focus: the \textbf{fit}, \textbf{LOO-CV}, model comparison, and
  \textbf{honest validation}
\end{itemize}

\vspace{0.4em}

\textbf{Thesis.} The prior buys sign-consistency and interpretability,
not predictive accuracy. What is validated independently of the data is
the \textbf{cross-feeding direction only} (below).
\end{frame}

Early peri-implant colonization. Each sample is treated as a composition

\end{frame}

\begin{frame}{Model --- gLV and replicator}
\protect\hypertarget{model-glv-and-replicator}{}
Generalized Lotka--Volterra for absolute abundances \(x_i\):
\[\dot{x}_i = x_i\Big(b_i + \sum_{j=1}^{S} A_{ij}\,x_j\Big),\qquad S=10 .\]

On the simplex \(\varphi_i=x_i/\sum_k x_k\), the replicator / Hamilton
form:
\[\dot{\varphi}_i = \varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big],
\qquad \sum_i\varphi_i=1 .\]

\begin{itemize}
\tightlist
\item
  \textbf{Hamilton} uses a \textbf{symmetric} \(A\) (replicator with
  linear payoffs; Taylor \& Jonker 1978).
\item
  \textbf{Classic gLV} uses an \textbf{asymmetric} \(A\). \(A_{ij}>0\):
  facilitation; \(A_{ij}<0\): suppression.
\end{itemize}
\end{frame}

\begin{frame}{Inference is underdetermined \(\Rightarrow\) sign prior}
\protect\hypertarget{inference-is-underdetermined-rightarrow-sign-prior}{}
Parameters \(\mathcal{O}(S^2)\approx100\) vs \(10\times3\) observations.
Abundance alone cannot identify the signs of \(A\) \(\Rightarrow\) a
one-sided hinge penalises \textbf{sign violations only}:
\[\mathcal{L}(A,b)=\frac{1}{2\sigma^{2}}\sum_{t}\big\lVert\varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}\big\rVert^{2}
\;+\; W\!\!\sum_{(i,j)\in\mathcal{M}}\!\!\left[-\,P_{ij}\,A_{ij}\right]_{+} .\]

\begin{itemize}
\tightlist
\item
  \(P_{ij}=\operatorname{sgn}(F_{ij})\) comes from AGORA cross-feeding
  flux (derived in the AGORA deck).
\item
  \(\left[-P_{ij}A_{ij}\right]_{+}\) is \(0\) when
  \(\operatorname{sgn}(A_{ij})=P_{ij}\) --- \textbf{magnitude is free}.
\item
  Posterior sampled by \textbf{TMCMC} (\(10^{4}\) particles).
\end{itemize}
\end{frame}

\begin{frame}{LOO-CV definition}
\protect\hypertarget{loo-cv-definition}{}
Re-fit with patient \(p\) held out, evaluate on its trajectory:
\[\mathrm{RMSE}=\sqrt{\frac{1}{NT}\sum_{t}\big\lVert\varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}\big\rVert^{2}},\]
\[\mathrm{BC}=\frac{\sum_i\lvert\varphi^{\text{obs}}_i-\varphi^{\text{pred}}_i\rvert}{\sum_i\big(\varphi^{\text{obs}}_i+\varphi^{\text{pred}}_i\big)}
\quad(\text{Bray–Curtis dissimilarity}).\]

Every one of the 10 patients is held out in turn (leave-one-patient-out)
and averaged.
\end{frame}

\begin{frame}{Honest interpretation}
\protect\hypertarget{honest-interpretation}{}
\begin{itemize}
\tightlist
\item
  The \textbf{AGORA prior itself is not reproduced} by the 16S dynamics
  \(\Rightarrow\) it is a \textbf{modelling choice}, not a
  data-confirmed fact.
\item
  What is validated independently of the data:

  \begin{enumerate}
  \tightlist
  \item
    the \textbf{cross-feeding direction} (Hamilton,
    \(p=4\times10^{-4}\)),
  \item
    \textbf{two-cohort strong-pair} signs (Dieckow \(\times\) Botelho,
    \(89\%\)).
  \end{enumerate}
\item
  Competition direction, prior magnitude, and the full matrix are not
  validated.
\end{itemize}

\vspace{0.3em}

We do not oversell: a \textbf{subset} of signs is supported; the rest
remains hypothesis.
\end{frame}

\begin{frame}{Conclusions}
\protect\hypertarget{conclusions}{}
\begin{enumerate}
\tightlist
\item
  Compositional 16S time-series \(\to\) a signed \(A\), via gLV /
  Hamilton replicator.
\item
  Under-determination is eased by the \textbf{AGORA metabolic sign
  prior} (one-sided hinge); posterior via TMCMC.
\item
  Best: L1+L2+AGORA \(W=1.0\) gives \textbf{LOO-RMSE \(0.0504\)}, SA
  \(100\%\). But prior-free gLV reaches \(0.0455\) --- the prior buys
  \textbf{interpretability}, not accuracy.
\item
  \textbf{Signs are stable under LOO} (\(\geq0.70\)); the MacArthur
  magnitude prior fails.
\item
  Honest validation: only the \textbf{cross-feeding direction}
  (\(p=4\times10^{-4}\)) and \textbf{two-cohort strong pairs} (\(89\%\))
  are independently supported. Clinical validation is in progress.
\end{enumerate}
\end{frame}

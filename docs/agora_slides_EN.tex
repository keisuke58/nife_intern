\begin{frame}{Scope and claim}
\protect\hypertarget{scope-and-claim}{}
Constrain an \textbf{ecological interaction matrix} with
\textbf{metabolism computed from genomes (AGORA2)}, and quantify how far
that constraint is empirically supported.

\begin{itemize}
\tightlist
\item
  the ecological model: generalized Lotka--Volterra (gLV) / replicator
\item
  the metabolic engine: flux-balance analysis (FBA, pFBA, MICOM)
\item
  the bridge: a \textbf{sign prior}
  \(P_{ij}=\operatorname{sgn}(F_{ij})\) derived from cross-feeding flux
\item
  the validation: permutation test on the prior-free fit, two-cohort
  replication
\end{itemize}

\vspace{0.4em}

\begin{block}{Claim}
The metabolic signal constrains the \textbf{sign} of cooperative
interactions (not the magnitude, not competition), supported at
\alert{$p=4\times10^{-4}$}.
\end{block}
\end{frame}

\begin{frame}{The ecological model}
\protect\hypertarget{the-ecological-model}{}
Generalized Lotka--Volterra for absolute abundances \(x_i\):
\[\dot{x}_i = x_i\Big(b_i + \sum_{j=1}^{S} A_{ij}\,x_j\Big),\qquad i=1,\dots,S=10 .\]

On the simplex \(\varphi_i = x_i/\sum_k x_k\) (compositional 16S data),
the replicator / Hamilton form is
\[\dot{\varphi}_i = \varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big],
\qquad \sum_i \varphi_i = 1 .\]

\begin{itemize}
\tightlist
\item
  \(A_{ij}>0\): \(j\) \textbf{promotes} \(i\); \(A_{ij}<0\): \(j\)
  \textbf{suppresses} \(i\).
\item
  Inference is \textbf{underdetermined}: \(\mathcal{O}(S^2)=100\)
  parameters vs \(10\) patients \(\times\,3\) weeks. The signs of \(A\)
  are not identifiable from abundance alone \(\Rightarrow\) we add a
  \textbf{metabolic sign prior}.
\end{itemize}
\end{frame}

\begin{frame}{Flux-balance analysis (FBA) as a linear program}
\protect\hypertarget{flux-balance-analysis-fba-as-a-linear-program}{}
A genome-scale model gives a stoichiometric matrix
\(S\in\mathbb{R}^{m\times n}\) (metabolites \(\times\) reactions).
Steady-state growth is the LP
\[\max_{v}\; c^{\!\top} v \quad\text{s.t.}\quad S v = 0,\;\; v^{-}\le v \le v^{+},\]
with \(c\) selecting the biomass reaction (\(\mu = c^{\!\top}v\)).
\textbf{Parsimonious FBA} removes flux loops by a second stage,
\[\min_{v}\; \sum_{j}\lvert v_j\rvert \quad\text{s.t.}\quad c^{\!\top}v=\mu^{\star},\; Sv=0,\; v^{-}\le v\le v^{+}.\]

Exchange fluxes give per-guild \textbf{secretion}
\(s_{j\alpha}=\left[v^{\text{ex}}_{j\alpha}\right]_{+}\) and
\textbf{uptake}
\(u_{i\alpha}=\left[-v^{\text{ex}}_{i\alpha}\right]_{+}\) over
metabolites \(\alpha\).
\end{frame}

\begin{frame}{What AGORA2 supplies}
\protect\hypertarget{what-agora2-supplies}{}
\textbf{AGORA2} (Heinken et al., \emph{Nat. Biotechnol.} 2023, 41:1320):
\textbf{7,302} genome-scale reconstructions, extended from gut to
\textbf{oral} taxa; one representative SBML model per guild.

\vspace{0.5em}
\begin{itemize}
\tightlist
\item
  representatives: Bacilli = \emph{S. gordonii}, Negativicutes =
  \emph{V. parvula}, Bacteroidia = \emph{P. melaninogenica},
  Fusobacteriia = \emph{F. nucleatum}, \ldots{}
\item
  oral-fluid medium (Dawes 2008): sugars, 20 amino acids, B-vitamins,
  trace metals, cell-wall precursors \(\Rightarrow\) positive growth for
  all 10 guilds (\(\mu = 0.11\text{–}1.66\,\mathrm{h^{-1}}\)).
\end{itemize}

\vspace{0.5em}

Medium design is a genuine sensitivity: a too-poor medium drives
\(\mu\to0\) and collapses the cross-feeding signal.
\end{frame}

\begin{frame}{Cross-feeding score \(\to\) sign prior}
\protect\hypertarget{cross-feeding-score-to-sign-prior}{}
Let \(\mathcal{T}=\{\text{H}_2\text{O}_2,\text{H}_2\text{S}\}\) be
toxins. Define a directed \textbf{net metabolic flow} from \(j\) to
\(i\),
\[F_{ij} \;=\; \underbrace{\sum_{\alpha\notin\mathcal{T}} w_\alpha\, s_{j\alpha}\,u_{i\alpha}}_{\text{cross-feeding }(+)}
\;-\; \underbrace{\sum_{\alpha\in\mathcal{T}} w_\alpha\, s_{j\alpha}\,u_{i\alpha}}_{\text{toxin }(-)} .\]

The \textbf{sign prior} keeps only the direction:
\[P_{ij} = \operatorname{sgn}(F_{ij}) \in \{-1,\,0,\,+1\}.\]

Magnitude is discarded by design --- FBA flux units \(\neq\) ecological
units. Across \(10\times 9=90\) directed pairs this yields the
constrained set \(\mathcal{M}=\{(i,j): P_{ij}\neq 0\}\).
\end{frame}

\begin{frame}{Bayesian integration: the sign-prior penalty}
\protect\hypertarget{bayesian-integration-the-sign-prior-penalty}{}
The fit minimises trajectory error plus a one-sided hinge that penalises
only \textbf{sign violations} (never magnitude):
\[\mathcal{L}(A,b) = \frac{1}{2\sigma^{2}}\sum_{t}\big\lVert \varphi^{\text{obs}}_{t}-\varphi^{\text{pred}}_{t}(A,b)\big\rVert^{2}
\;+\; W\!\!\sum_{(i,j)\in\mathcal{M}}\!\! \left[-\,P_{ij}\,A_{ij}\right]_{+} .\]

\begin{itemize}
\tightlist
\item
  \(\left[-P_{ij}A_{ij}\right]_{+}=\max(0,-P_{ij}A_{ij})\) is \(0\) when
  \(\operatorname{sgn}(A_{ij})=P_{ij}\).
\item
  \(W\) = prior stiffness; \(\sigma\) = observation scale.
\item
  Posterior sampled by \textbf{TMCMC} (\(10^4\) particles) for the
  5-species attractors; guild-level fits by L-BFGS-B / TMCMC.
\end{itemize}
\end{frame}

\begin{frame}{Why magnitude priors fail --- the MacArthur view}
\protect\hypertarget{why-magnitude-priors-fail-the-macarthur-view}{}
Consumer--resource theory (MacArthur 1970; Marsland 2019) derives
\[A_{ij} = \underbrace{\sum_{\alpha} s_{j\alpha}\,c_{i\alpha}}_{\text{cross-feeding}\,(+)}
\;-\; \underbrace{\frac{c_i\!\cdot\! c_j}{\lVert c_i\rVert\,\lVert c_j\rVert}}_{\text{niche overlap}\,(-)} .\]

\vspace{0.5em}
\begin{itemize}
\tightlist
\item
  \textbf{Niche-overlap (cosine) term saturates}: oral taxa are
  generalists, so \(\cos(c_i,c_j)\approx1\) for nearly all pairs
  \(\Rightarrow\) everything reads as competition (empirically \(6\)
  positive / \(84\) negative pairs --- useless).
\item
  Growth-rate-suppression priors fail identically in poor media.
\end{itemize}

\vspace{0.5em}
\(\Rightarrow\) keep \(\operatorname{sgn}(A_{ij})\), discard the
magnitude.
\end{frame}

\begin{frame}{MICOM --- community FBA}
\protect\hypertarget{micom-community-fba}{}
Single-species pFBA tests only \textbf{feasibility} (\(j\) \emph{can}
secrete \(X\), \(i\) \emph{can} eat \(X\)). \textbf{MICOM} (Diener 2020)
instead solves the joint community and checks whether the flux
\emph{actually flows}, via a \textbf{cooperative trade-off}:
\[\max\; \min_{i}\frac{\mu_i}{\mu_i^{\max}}\quad\text{s.t.}\quad
\sum_i \mu_i \ge \tau\sum_i \mu_i^{\max},\;\; S^{\text{com}}v=0,\;\tau=0.5 .\]

\vspace{0.5em}
\begin{itemize}
\tightlist
\item
  each guild is guaranteed at least a fraction \(\tau\) of its maximal
  growth, then the shared flux pattern is fixed;
\item
  even though oral taxa are generalists, \textbf{only routes carrying
  real community flux activate} --- the unspecific ``everyone competes''
  artefact disappears;
\item
  the secretion/uptake exchange fluxes \(s_{j\alpha},u_{i\alpha}\) are
  read off the community solution and fed into \(F_{ij}\) exactly as
  before.
\end{itemize}
\end{frame}

\begin{itemize}
\tightlist
\item
  \textbf{Only the cross-feeding direction is validated}; competition is

\begin{frame}{Limitations and conclusions}
\protect\hypertarget{limitations-and-conclusions}{}
\textit{Guild $=$ class (representative $\neq$ guild); $20/22$ inhibition rows are oxygen (no producer) $\Rightarrow$ only the H$_2$O$_2$ toxin fires; $w_\alpha$ is a per-metabolite max; magnitude is discarded.}

\vspace{0.4em}
\textbf{Conclusions:}
\begin{enumerate}
\tightlist
\item \(\text{AGORA pFBA}\to F_{ij}\to P_{ij}=\operatorname{sgn}(F_{ij})\) is the methodological novelty.
\item \textbf{Sign usable, magnitude not} (MacArthur cosine saturation avoided).
\item Single-species (\(92\%\)) \(\to\) \textbf{MICOM (\(100\%\))} captures community context.
\item Honest validation: \textbf{cross-feeding only}, \(p=4\times10^{-4}\); two cohorts \(89\%\); the prior is a modelling choice.
\item COMETS dFBA reproduces the dysbiotic split forward.
\end{enumerate}
\end{frame}

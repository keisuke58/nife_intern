\begin{frame}{Question: adding WHERE to WHO}
\protect\hypertarget{question-adding-where-to-who}{}
The bulk ODE fit gives \textbf{who interacts with whom (WHO)}; the FISH
depth profiles add \textbf{where that interaction happens (WHERE)}.

\begin{itemize}
\tightlist
\item
  Known: bulk gLV/Hamilton interaction matrix \(A\) and growth vector
  \(b\) (Dieckow fit).
\item
  Unknown: per-species \textbf{transport} along the substratum depth
  axis.
\item
  Goal: with the reaction term fixed from the bulk fit, infer
  \textbf{only the spatial transport parameters} from the depth-resolved
  FISH data.
\end{itemize}

\vspace{0.4em}

\textbf{Thesis.} Dysbiosis manifests not as a change in bulk composition
but as a \textbf{spatial reorganisation} --- in particular \emph{P.
gingivalis} sinks deep and decouples from \emph{F. nucleatum}.
\end{frame}

\begin{frame}{Governing equation (reaction--diffusion PDE)}
\protect\hypertarget{governing-equation-reactiondiffusion-pde}{}
For depth \(z\in[0,L]\) and composition \(\varphi_i(z,t)\), the
reaction--advection--diffusion equation is
\[\frac{\partial \varphi_i}{\partial t}
 = D_i\,\frac{\partial^2 \varphi_i}{\partial z^2}
 \;-\; u\,\frac{\partial \varphi_i}{\partial z}
 \;+\; R_i(\varphi).\]

\begin{itemize}
\tightlist
\item
  \(R_i(\varphi)\) is the replicator / Hamilton reaction; \(A,b\) are
  \textbf{fixed} from the bulk fit,
  \[R_i(\varphi)=\varphi_i\Big[(A\varphi+b)_i - \varphi^{\!\top}(A\varphi+b)\Big]+\gamma\,\varphi_i.\]
\item
  The \textbf{only free parameters are the diffusivities \(D_i\) and the
  advection \(u\)}.
\item
  \(\gamma\) is a Lagrange term enforcing
  \(\sum_i \varphi_i + \varphi_0 = 1\) at each \(z\).
\end{itemize}
\end{frame}

\begin{frame}{Boundary conditions and numerics}
\protect\hypertarget{boundary-conditions-and-numerics}{}
\textbf{Boundary conditions:}
\[\left.\frac{\partial \varphi_i}{\partial z}\right|_{z=0}=0\quad(\text{substratum, no-flux}),\qquad
\varphi_i\big|_{z=L}=\varphi_{\text{bulk},i}\quad(\text{bulk, Dirichlet}).\]
\(\varphi_{\text{bulk},i}\) is fixed to the Day-1 median.

\textbf{Numerics (Method of Lines):} Lie operator splitting alternates
reaction and transport,
\[e^{\Delta t\,\mathcal{L}} \approx e^{\Delta t\,\mathcal{R}}\,e^{\Delta t\,\mathcal{T}} + \mathcal{O}(\Delta t).\]

\begin{itemize}
\tightlist
\item
  Reaction \(\mathcal{R}\): implicit Euler with Newton iteration per
  \(z\)-node (stiff).
\item
  Transport \(\mathcal{T}\): explicit finite difference (central
  \(\partial_z^2\), upwind \(\partial_z\)).
\end{itemize}
\end{frame}

\begin{frame}{Inverse problem}
\protect\hypertarget{inverse-problem}{}
Least-squares estimation of the transport parameters against the
observation \(\varphi^{\text{obs}}(z,t)\):
\[\min_{D,\,u}\;\; \sum_t \sum_z
 \big\lVert \varphi^{\text{pred}}(z,t;D,u) - \varphi^{\text{obs}}(z,t)\big\rVert^2 .\]

\begin{itemize}
\tightlist
\item
  Optimiser: \textbf{L-BFGS-B}, with a numerical gradient through the
  stiff PDE (JAX forward model).
\item
  The forward model is the MoL solver above; with \(A,b\) fixed the
  unknowns are \(D\in\mathbb{R}^5\) and \(u\).
\item
  Non-dimensionalisation: the \textbf{Péclet number}
  \(\mathrm{Pe} = uL/D_i\) measures advection vs diffusion
  (\(\mathrm{Pe}\gg1\) advection-dominated, \(\mathrm{Pe}\ll1\)
  diffusion-dominated).
\end{itemize}
\end{frame}

\begin{frame}{Other: bulk CH ≈ DH}
\protect\hypertarget{other-bulk-ch-dh}{}
The CH--DH \textbf{Bray--Curtis divergence} is \(\approx 0.2\) and
\textbf{flat from Day 1}.

\begin{itemize}
\tightlist
\item
  The difference is set by the inoculum and barely grows over time.
\item
  In bulk composition CH \(\approx\) DH \(\Rightarrow\) dysbiosis is
  \textbf{spatial / temporal}, not a shift in bulk composition.
\end{itemize}

\vspace{0.4em}

This is consistent with findings 1--2: the difference is not ``who and
how much'' but ``where they sit and what they partner with''.
\end{frame}

\begin{frame}{Conclusions}
\protect\hypertarget{conclusions}{}
\begin{enumerate}
\tightlist
\item
  We build a framework that adds FISH depth profiles (WHERE) to the bulk
  ODE (WHO) and, with the reaction term fixed, infers \textbf{only the
  transport parameters \(D_i,u\)}.
\item
  \textbf{Dysbiosis \(=\) spatial reorganisation}: \emph{P. gingivalis}
  sinks deep (\(+30\,\mu\mathrm{m}\)) and goes autonomous from \emph{F.
  nucleatum} (Manders \(M_1\) high only early).
\item
  Bulk is CH \(\approx\) DH (Bray--Curtis \(\approx0.2\), flat) --- the
  difference is spatial, not bulk.
\item
  ``Central species move less'' (\(D\) anti-correlates with centrality);
  the in-vitro↔in-vivo rank order agrees (\(\rho=0.70\)).
\item
  \textcolor{red}{Transport parameters are **preliminary** (not converged)};
  a sweep with faster settings is converging on the HPC.
\end{enumerate}
\end{frame}

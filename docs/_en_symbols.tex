% EN builds use a Times-family text font (STIX Two Text) that lacks a few set /
% relation glyphs. Route those Unicode symbols through math mode so they always
% render (regardless of the text/mono font), keeping clean Times for the prose.
\usepackage{newunicodechar}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{↔}{\ensuremath{\leftrightarrow}}
\newunicodechar{↑}{\ensuremath{\uparrow}}
\newunicodechar{∩}{\ensuremath{\cap}}
\newunicodechar{−}{\ensuremath{-}}
\newunicodechar{≠}{\ensuremath{\neq}}
\newunicodechar{×}{\ensuremath{\times}}
\newunicodechar{≈}{\ensuremath{\approx}}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{≤}{\ensuremath{\leq}}
